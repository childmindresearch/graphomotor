129class GridCell:
130    """Represents a single rectangular region in a grid layout.
131
132    Used to assign strokes to letter regions (Alphabet) or digit regions (DSYM).
133    Boundary policy: uses half-open intervals [min, max) to prevent
134    double-assignment on shared grid edges. The outer grid boundaries should be
135    padded (e.g., by 0.1) at the Grid level so that centroids on the outermost
136    edge are not excluded.
137
138    Attributes:
139        x_min: Left boundary of the cell.
140        x_max: Right boundary of the cell.
141        y_min: Bottom boundary of the cell.
142        y_max: Top boundary of the cell.
143        index: Position of the cell in the grid (0-based).
144        label: Display label for the cell (e.g., 'A', 'B', '1'). Defaults to an
145            empty string, which marks the cell as unlabeled: either a purely
146            spatial grid where cells are identified by index rather than name,
147            or a cell that is not of interest for the analysis (e.g., a spacer
148            or ignored region that no stroke should be attributed to).
149        strokes: List of Stroke objects assigned to this cell.
150    """
151
152    x_min: float
153    x_max: float
154    y_min: float
155    y_max: float
156    index: int = 0
157    label: str = ""
158    strokes: List["Stroke"] = dataclasses.field(default_factory=list)
159
160    def __post_init__(self) -> None:
161        """Validate that min bounds are strictly less than max bounds.
162
163        Raises:
164            ValueError: If x_min >= x_max or y_min >= y_max.
165        """
166        if self.x_min >= self.x_max:
167            raise ValueError(
168                f"x_min ({self.x_min}) must be less than x_max ({self.x_max})"
169            )
170        if self.y_min >= self.y_max:
171            raise ValueError(
172                f"y_min ({self.y_min}) must be less than y_max ({self.y_max})"
173            )
174
175    def contains_points(self, points: pd.DataFrame) -> bool:
176        """Check if a stroke belongs to this cell based on its centroid.
177
178        Computes the centroid (mean x, mean y) of the provided points and checks
179        whether it falls within the cell boundaries. Uses half-open intervals
180        [min, max) to prevent double-assignment on shared grid edges.
181
182        Args:
183            points: DataFrame with 'x' and 'y' columns representing a stroke.
184
185        Returns:
186            True if the stroke centroid is within the cell, False otherwise.
187        """
188        centroid_x = points["x"].mean()
189        centroid_y = points["y"].mean()
190        return (
191            self.x_min <= centroid_x < self.x_max
192            and self.y_min <= centroid_y < self.y_max
193        )

197class Grid:
198    """Represents a rectangular grid composed of multiple GridCell objects.
199
200    The grid divides a bounding box into rows and columns, where each cell can
201    hold strokes assigned by centroid location. Cells are ordered left-to-right,
202    top-to-bottom (row-major order).
203
204    Attributes:
205        cells: List of GridCell objects that compose the grid.
206    """
207
208    cells: List[GridCell] = dataclasses.field(default_factory=list)
209
210    @classmethod
211    def from_bbox(
212        cls,
213        x_min: float,
214        x_max: float,
215        y_min: float,
216        y_max: float,
217        n_rows: int,
218        n_cols: int,
219        labels: Optional[List[str]] = None,
220        padding: float = 0.1,
221    ) -> "Grid":
222        """Create a Grid by subdividing a bounding box into rows and columns.
223
224        Cells are generated in row-major order (left-to-right, top-to-bottom).
225        Interior cell boundaries are exact subdivisions of the bounding box;
226        only the outermost edges are extended by ``padding`` so that centroids
227        falling exactly on the outer boundary are still captured by a cell.
228
229        Args:
230            x_min: Left boundary of the bounding box.
231            x_max: Right boundary of the bounding box.
232            y_min: Bottom boundary of the bounding box.
233            y_max: Top boundary of the bounding box.
234            n_rows: Number of rows in the grid.
235            n_cols: Number of columns in the grid.
236            labels: Optional list of labels for each cell, assigned in
237                row-major order. Must have length n_rows * n_cols if provided.
238                If omitted, every cell is left with an empty-string label,
239                marking the grid as unlabeled (cells identified by index only).
240                An empty label also denotes a cell that is not of interest for
241                the analysis (e.g., a spacer or ignored region).
242            padding: Amount to extend the outermost cell edges to capture edge
243                centroids; interior boundaries are unaffected (default 0.1).
244
245        Returns:
246            A Grid instance populated with GridCell objects.
247
248        Raises:
249            ValueError: If n_rows or n_cols is less than 1, or if the length
250                of labels does not match n_rows * n_cols.
251        """
252        if n_rows < 1 or n_cols < 1:
253            raise ValueError("n_rows and n_cols must be at least 1.")
254        n_cells = n_rows * n_cols
255        if labels is not None and len(labels) != n_cells:
256            raise ValueError(
257                f"labels length ({len(labels)}) must match n_rows * n_cols ({n_cells})."
258            )
259
260        col_width = (x_max - x_min) / n_cols
261        row_height = (y_max - y_min) / n_rows
262
263        cells: List[GridCell] = []
264        index = 0
265        for row in range(n_rows):
266            for col in range(n_cols):
267                cell_x_min = x_min + col * col_width
268                if col == 0:
269                    cell_x_min -= padding
270
271                cell_x_max = x_min + (col + 1) * col_width
272                if col == n_cols - 1:
273                    cell_x_max += padding
274
275                cell_y_min = y_max - (row + 1) * row_height
276                if row == n_rows - 1:
277                    cell_y_min -= padding
278
279                cell_y_max = y_max - row * row_height
280                if row == 0:
281                    cell_y_max += padding
282
283                label = labels[index] if labels is not None else ""
284                cells.append(
285                    GridCell(
286                        x_min=cell_x_min,
287                        x_max=cell_x_max,
288                        y_min=cell_y_min,
289                        y_max=cell_y_max,
290                        index=index,
291                        label=label,
292                    )
293                )
294                index += 1
295
296        return cls(cells=cells)

298    def get_cell_for_point(self, x: float, y: float) -> int:
299        """Return the index of the cell containing the given point.
300
301        Iterates through cells and returns the index of the first cell whose
302        half-open interval [min, max) contains the point. Returns -1 if no
303        cell contains the point.
304
305        Args:
306            x: X coordinate of the point.
307            y: Y coordinate of the point.
308
309        Returns:
310            The index of the matching cell, or -1 if no cell contains the point.
311        """
312        for cell in self.cells:
313            if cell.x_min <= x < cell.x_max and cell.y_min <= y < cell.y_max:
314                return cell.index
315        return -1

432    def valid_ink_trajectory(
433        self,
434        start_circle: CircleTarget,
435        end_circle: CircleTarget,
436    ) -> Tuple[Optional[int], Optional[int]]:
437        """Determine whether an ink trajectory exists from a start to end circle.
438
439        An "ink trajectory" is defined as the first contiguous sequence of
440        points that:
441        1. Begins **after** the pen leaves the start circle, and
442        2. Ends when the pen first enters the end circle.
443
444        The function scans point-by-point in order. The ink start index is the
445        first point whose (x, y) location is *outside* the start circle. The
446        ink end index is the first subsequent point whose (x, y) location falls
447        *inside* the end circle. If either of these conditions never occurs,
448        the trajectory is considered invalid. If a valid trajectory is found,
449        the ink_points attribute is updated to contain only the points within
450        this trajectory.
451
452        Args:
453            points: DataFrame of points with 'x' and 'y' columns.
454            start_circle: CircleTarget representing the start circle.
455            end_circle: CircleTarget representing the end circle.
456
457        Returns:
458            Tuple of (ink_start_idx: Optional[int], ink_end_idx: Optional[int]) if valid
459            trajectory exists, else (None, None).
460        """
461        ink_start_idx = None
462        ink_end_idx = None
463
464        for idx, row in self.points.iterrows():
465            if (
466                not start_circle.contains_point(row["x"], row["y"])
467                and ink_start_idx is None
468            ):
469                ink_start_idx = idx
470
471            if ink_start_idx is not None and end_circle.contains_point(
472                row["x"], row["y"]
473            ):
474                ink_end_idx = idx
475                break
476
477        return ink_start_idx, ink_end_idx

479    def calculate_path_optimality(
480        self,
481        start_circle: CircleTarget,
482        end_circle: CircleTarget,
483    ) -> None:
484        """Calculate path optimality ratio.
485
486        The default value for path optimality in the LineSegment object is 0.0. This
487        function updates the path_optimality attribute of the LineSegment object based
488        on the optimal distance between the start and end circles, adjusted for their
489        radii. If the optimal distance is less than or equal to zero, the path
490        optimality remains 0.0.
491
492        Args:
493            segment: LineSegment object for which to calculate path optimality.
494            start_circle: CircleTarget representing the start circle.
495            end_circle: CircleTarget representing the end circle.
496
497        Returns:
498            Path optimality ratio.
499        """
500        optimal_distance = (
501            dist.euclidean(
502                [start_circle.center_x, start_circle.center_y],
503                [end_circle.center_x, end_circle.center_y],
504            )
505            - start_circle.radius
506            - end_circle.radius
507        )
508
509        if optimal_distance > 0:
510            self.path_optimality = optimal_distance / self.distance
511        return

513    def calculate_velocity_metrics(self) -> None:
514        """Get velocity metrics of a LineSegment.
515
516        Args:
517            self: LineSegment object to calculate velocities for.
518        """
519        dx = np.diff(self.ink_points["x"].values)
520        dy = np.diff(self.ink_points["y"].values)
521        dt = np.diff(self.ink_points["seconds"].values)
522
523        distances = np.sqrt(dx**2 + dy**2)
524        self.distance = np.sum(distances)
525
526        velocities = distances / dt
527        self.velocities = velocities.tolist()
528
529        self.mean_speed = np.mean(velocities)
530        self.speed_variance = np.var(velocities)
531
532        if len(velocities) >= 2:
533            self.accelerations = np.diff(velocities).tolist()
534
535        return

537    def detect_hesitations(self, threshold_percentile: int = 20) -> None:
538        """Detect hesitations as periods of significantly reduced velocity.
539
540        This function defines a hesitation as any period where the velocity falls below
541        a certain threshold, which is determined by the specified percentile of the
542        velocity distribution. It counts the number of distinct hesitation periods and
543        adds 1 if the line starts with a hesitation. It also calculates the total
544        duration of hesitations based on the number of points that fall below the
545        threshold and the time interval between points.
546
547        hesitation_count defaults to 0 and hesitation_duration defaults to 0.0 in the
548        LineSegment object if there are less than 3 velocity points. This function also
549        assumes uniform sampling.
550
551        Args:
552            threshold_percentile: Percentile to determine the velocity threshold for
553                hesitations (default is 20, meaning the bottom 20% of velocities are
554                considered hesitations).
555        """
556        if len(self.velocities) < 3:
557            return
558
559        dt = np.diff(self.ink_points["seconds"].values)
560
561        threshold_velocity = np.percentile(self.velocities, threshold_percentile)
562        hesitations = self.velocities < threshold_velocity
563
564        hesitation_changes = np.diff(hesitations.astype(int))
565        hesitation_count = np.sum(hesitation_changes == 1)
566
567        if hesitations[0]:
568            hesitation_count += 1
569
570        self.hesitation_count = hesitation_count
571        self.hesitation_duration = np.sum(hesitations) * dt[0]
572
573        return

575    def calculate_smoothness(self) -> None:
576        """Calculate path smoothness based on Root Mean Square (RMS) curvature.
577
578        Represents the curvature per unit arc length.
579        Lower values indicate smoother drawings. Penalizes sharp corners (e.g.,
580        90° turns) and noisy corrections. Normalized by arc length to reduce
581        sampling-rate dependence.
582        """
583        if len(self.ink_points) < 3:
584            return
585
586        xy = self.ink_points[["x", "y"]].to_numpy()
587
588        forward_vector = xy[1:-1] - xy[:-2]
589        backward_vector = xy[2:] - xy[1:-1]
590
591        forward_norm = np.linalg.norm(forward_vector, axis=1)
592        backward_norm = np.linalg.norm(backward_vector, axis=1)
593
594        valid = (forward_norm > 0) & (backward_norm > 0)
595        if not np.any(valid):
596            return
597
598        valid_forward_vector = forward_vector[valid]
599        valid_backward_vector = backward_vector[valid]
600        valid_forward_norm = forward_norm[valid]
601        valid_backward_norm = backward_norm[valid]
602
603        cos_angle = (valid_forward_vector * valid_backward_vector).sum(axis=1) / (
604            valid_forward_norm * valid_backward_norm
605        )
606        cos_angle = np.clip(cos_angle, -1.0, 1.0)
607
608        angles = np.arccos(cos_angle)
609
610        avg_segment_length = (valid_forward_norm + valid_backward_norm) / 2.0
611        curvatures = angles / avg_segment_length
612
613        self.smoothness = float(np.sqrt(np.mean(curvatures**2)))
614
615        return

617    def compute_segment_metrics(
618        self, circles: dict[str, dict[str, CircleTarget]], trail_id: str
619    ) -> None:
620        """Compute all metrics for a line segment.
621
622        This function computes various metrics for the line segment, including ink time,
623        velocity metrics, path optimality, smoothness, and hesitation detection. It
624        first determines the valid ink trajectory between the start and end circles. If
625        a valid trajectory is found, it updates the ink_points attribute and calculates
626        the metrics.
627
628        Args:
629            circles: A dictionary mapping each trail type to dictionaries of
630                CircleTarget instances (output of load_scaled_circles in config).
631            trail_id: Trail identifier for circle lookup.
632        """
633        from graphomotor.core import config
634
635        logger = config.get_logger()
636        trail_circles = circles[trail_id]
637        points = self.points.copy()
638
639        if len(points) < 2:
640            logger.warning(
641                "Not enough points to calculate metrics for line segment: "
642                "start=%s end=%s",
643                self.start_label,
644                self.end_label,
645            )
646            return
647
648        if self.start_label not in trail_circles or self.end_label not in trail_circles:
649            logger.warning(

538    def detect_hesitations(self, threshold_percentile: int = 20) -> None:
539        """Detect hesitations as periods of significantly reduced velocity.
540
541        This function defines a hesitation as any period where the velocity falls below
542        a certain threshold, which is determined by the specified percentile of the
543        velocity distribution. It counts the number of distinct hesitation periods and
544        adds 1 if the line starts with a hesitation. It also calculates the total
545        duration of hesitations based on the number of points that fall below the
546        threshold and the time interval between points.

549        LineSegment object if there are less than 3 velocity points. This function also

554                hesitations (default is 20, meaning the bottom 20% of velocities are
555                considered hesitations).
556        """

558            return

566        hesitation_count = np.sum(hesitation_changes == 1)
567

576    def calculate_smoothness(self) -> None:
577        """Calculate path smoothness based on Root Mean Square (RMS) curvature.
578
579        Represents the curvature per unit arc length.
580        Lower values indicate smoother drawings. Penalizes sharp corners (e.g.,
581        90° turns) and noisy corrections. Normalized by arc length to reduce
582        sampling-rate dependence.
583        """
584        if len(self.ink_points) < 3:
585            return
586
587        xy = self.ink_points[["x", "y"]].to_numpy()
588
589        forward_vector = xy[1:-1] - xy[:-2]
590        backward_vector = xy[2:] - xy[1:-1]
591
592        forward_norm = np.linalg.norm(forward_vector, axis=1)
593        backward_norm = np.linalg.norm(backward_vector, axis=1)
594
595        valid = (forward_norm > 0) & (backward_norm > 0)
596        if not np.any(valid):
597            return
598
599        valid_forward_vector = forward_vector[valid]
600        valid_backward_vector = backward_vector[valid]
601        valid_forward_norm = forward_norm[valid]
602        valid_backward_norm = backward_norm[valid]
603
604        cos_angle = (valid_forward_vector * valid_backward_vector).sum(axis=1) / (
605            valid_forward_norm * valid_backward_norm
606        )
607        cos_angle = np.clip(cos_angle, -1.0, 1.0)
608
609        angles = np.arccos(cos_angle)
610
611        avg_segment_length = (valid_forward_norm + valid_backward_norm) / 2.0
612        curvatures = angles / avg_segment_length
613
614        self.smoothness = float(np.sqrt(np.mean(curvatures**2)))
615
616        return

577        """Calculate path smoothness based on Root Mean Square (RMS) curvature.
578
579        Represents the curvature per unit arc length.
580        Lower values indicate smoother drawings. Penalizes sharp corners (e.g.,
581        90° turns) and noisy corrections. Normalized by arc length to reduce
582        sampling-rate dependence.
583        """
584        if len(self.ink_points) < 3:
585            return
586
587        xy = self.ink_points[["x", "y"]].to_numpy()
588
589        forward_vector = xy[1:-1] - xy[:-2]
590        backward_vector = xy[2:] - xy[1:-1]
591
592        forward_norm = np.linalg.norm(forward_vector, axis=1)
593        backward_norm = np.linalg.norm(backward_vector, axis=1)
594
595        valid = (forward_norm > 0) & (backward_norm > 0)
596        if not np.any(valid):
597            return
598
599        valid_forward_vector = forward_vector[valid]
600        valid_backward_vector = backward_vector[valid]
601        valid_forward_norm = forward_norm[valid]
602        valid_backward_norm = backward_norm[valid]
603
604        cos_angle = (valid_forward_vector * valid_backward_vector).sum(axis=1) / (
605            valid_forward_norm * valid_backward_norm
606        )
607        cos_angle = np.clip(cos_angle, -1.0, 1.0)
608
609        angles = np.arccos(cos_angle)
610
611        avg_segment_length = (valid_forward_norm + valid_backward_norm) / 2.0
612        curvatures = angles / avg_segment_length
613
614        self.smoothness = float(np.sqrt(np.mean(curvatures**2)))
615
616        return
